# tidewick

A small, dependency-free Python library and command-line tool that predicts tide heights for a handful of
fictional stations from harmonic constituents stored as JSON. It exists as example material: the stations,
the numbers and the project are made up.

## Quick start

```bash
python -m tidewick stations                 # list the stations in the data directory
python -m tidewick predict port-avery --hours 24
python -m tidewick predict cape-mallow --at 2026-03-01T06:00
```

Python 3.11 or later. No third-party packages, by policy (see CONTRIBUTING.md).

All times in and out of the tool are a station's local **standard** time, year-round; tidewick never applies
a daylight-saving shift (see `docs/ARCHITECTURE.md`). A printed tide table that follows the local civil clock
will run an hour ahead of tidewick's times while daylight saving is in effect.

## Layout

- `tidewick/`: the package. `predict.py` does the arithmetic, `stations.py` loads station files,
  `harmonics.py` holds the constituent speeds, `cli.py` is the command line, `config.py` finds the data.
- `data/`: `stations.json` and one constituents file per station.
- `tests/`: the unit tests. Run them with `make test`.
- `tools/`: `check_style.py` (the style rules `make check` enforces) and `make_fixtures.py` (writes a
  constituents file for a new station).
- `docs/`: `ARCHITECTURE.md` and the decision records under `docs/adr/`.

## Data directory

The tool reads station data from the directory named in `TIDEWICK_DATA_DIR`, or `./data` when that is not
set.

"""The prediction arithmetic."""

import math
from datetime import datetime, timedelta

from tidewick import harmonics
from tidewick.stations import Station

EPOCH = datetime(2026, 1, 1)  # local standard time at every station

FEET_PER_METER = 3.28084


def to_units(meters: float, units: str) -> float:
    """Convert a height in metres to the given unit ('m' or 'ft')."""
    if units == "m":
        return meters
    if units == "ft":
        return meters * FEET_PER_METER
    raise ValueError(f"unknown units {units!r}; expected 'm' or 'ft'")


def height_at(station: Station, when: datetime) -> float:
    """Predicted height in metres at a local standard time."""
    t_hours = (when - EPOCH).total_seconds() / 3600.0
    total = station.mean_level_m
    for name, amplitude, phase_deg in station.constituents:
        angle = math.radians(harmonics.speed(name) * t_hours + phase_deg)
        total += amplitude * math.cos(angle)
    return total


def series(station: Station, start: datetime, hours: int, step_minutes: int = 60) -> list:
    """(time, height) pairs from start for the given number of hours."""
    if hours <= 0 or step_minutes <= 0:
        raise ValueError("hours and step_minutes must be positive")
    points = []
    steps = int(hours * 60 / step_minutes) + 1
    for i in range(steps):
        when = start + timedelta(minutes=i * step_minutes)
        points.append((when, height_at(station, when)))
    return points

"""Loading and validating station data files."""

import json
import re
from dataclasses import dataclass

from tidewick import config, harmonics

STATION_ID = re.compile(r"^[a-z0-9]+(-[a-z0-9]+)*$")


@dataclass(frozen=True)
class Station:
    id: str
    name: str
    mean_level_m: float
    utc_offset_hours: float
    datum: str
    constituents: tuple  # of (name, amplitude_m, phase_deg)


def list_ids() -> list:
    """The station ids listed in stations.json, in file order."""
    index = json.loads((config.data_dir() / "stations.json").read_text())
    return [entry["id"] for entry in index["stations"]]


def load(station_id: str) -> Station:
    """Load one station by id. Raises ValueError when the id is malformed or the files disagree."""
    if not STATION_ID.match(station_id):
        raise ValueError(f"station id {station_id!r} is not lowercase kebab-case")
    index = json.loads((config.data_dir() / "stations.json").read_text())
    entry = next((e for e in index["stations"] if e["id"] == station_id), None)
    if entry is None:
        raise ValueError(f"no station {station_id!r} in stations.json")
    path = config.data_dir() / "constituents" / f"{station_id}.json"
    doc = json.loads(path.read_text())
    if doc.get("station") != station_id:
        found = doc.get("station")
        raise ValueError(f"{path.name} says station {found!r}, expected {station_id!r}")
    constituents = []
    for c in doc["constituents"]:
        harmonics.speed(c["name"])  # validates the name
        constituents.append((c["name"], float(c["amplitude_m"]), float(c["phase_deg"])))
    return Station(
        id=station_id,
        name=entry["name"],
        mean_level_m=float(entry["mean_level_m"]),
        utc_offset_hours=float(entry["utc_offset_hours"]),
        datum=entry.get("datum", "MSL"),
        constituents=tuple(constituents),
    )

"""Command line: python -m tidewick {stations,predict}."""

import argparse
from datetime import datetime

from tidewick import __version__, predict, stations


def parse_args(argv=None):
    parser = argparse.ArgumentParser(prog="tidewick", description="Tide predictions.")
    parser.add_argument("--version", action="version", version=__version__)
    sub = parser.add_subparsers(dest="command", required=True)
    sub.add_parser("stations", help="list station ids")
    p = sub.add_parser("predict", help="predict heights for a station")
    p.add_argument("station")
    p.add_argument("--hours", type=int, default=24)
    p.add_argument("--step-minutes", type=int, default=60)
    p.add_argument(
        "--at",
        help=(
            "a single time, YYYY-MM-DDTHH:MM, in the station's local STANDARD time "
            "(no daylight-saving adjustment; see docs/ARCHITECTURE.md)"
        ),
    )
    p.add_argument("--units", choices=["m", "ft"], default="m", help="height units (default: m)")
    return parser.parse_args(argv)


def main(argv=None) -> int:
    args = parse_args(argv)
    if args.command == "stations":
        for station_id in stations.list_ids():
            print(station_id)
        return 0
    station = stations.load(args.station)
    if args.at:
        when = datetime.fromisoformat(args.at)
        height = predict.to_units(predict.height_at(station, when), args.units)
        print(
            f"{station.name} ({station.datum})  {when.isoformat(timespec='minutes')}  "
            f"{height:.2f} {args.units}"
        )
        return 0
    start = datetime.now().replace(minute=0, second=0, microsecond=0)
    for when, height in predict.series(station, start, args.hours, args.step_minutes):
        height = predict.to_units(height, args.units)
        print(f"{when.isoformat(timespec='minutes')}  {height:6.2f} {args.units}")
    return 0

import os
import unittest
from datetime import datetime

from tidewick import predict, stations


class PredictTest(unittest.TestCase):
    def test_height_is_near_mean_level_on_average(self):
        s = stations.load("port-avery")
        points = predict.series(s, datetime(2026, 3, 1), hours=48, step_minutes=30)
        mean = sum(h for _, h in points) / len(points)
        self.assertAlmostEqual(mean, s.mean_level_m, delta=0.15)

    def test_series_rejects_zero_hours(self):
        s = stations.load("cape-mallow")
        with self.assertRaises(ValueError):
            predict.series(s, datetime(2026, 3, 1), hours=0)

    def test_to_units_converts_meters_to_feet(self):
        self.assertAlmostEqual(predict.to_units(1.0, "ft"), 3.28084, places=5)
        self.assertEqual(predict.to_units(1.0, "m"), 1.0)

    def test_to_units_rejects_unknown_unit(self):
        with self.assertRaises(ValueError):
            predict.to_units(1.0, "cubit")

    @unittest.skipUnless(os.environ.get("TIDEWICK_SLOW") == "1", "set TIDEWICK_SLOW=1 to run")
    def test_year_long_series_has_two_tides_a_day(self):
        s = stations.load("port-avery")
        points = predict.series(s, datetime(2026, 1, 1), hours=24 * 365, step_minutes=20)
        highs = sum(
            1 for i in range(1, len(points) - 1) if points[i - 1][1] < points[i][1] > points[i + 1][1]
        )
        self.assertTrue(680 < highs < 720, highs)


if __name__ == "__main__":

 === docs/ARCHITECTURE.md ===
# Architecture

tidewick predicts a tide height h(t) at a station as

    h(t) = Z0 + sum over constituents of A_i * cos(speed_i * t_hours + phase_i)

where Z0 is the station's mean level, A_i and phase_i come from the station's constituents file, and speed_i
(degrees per hour) comes from `harmonics.SPEEDS`. t_hours is hours since the station's epoch,
2026-01-01T00:00 in the station's local standard time. Station files give times in local standard time all year:
there is no daylight-saving adjustment anywhere in the code, on purpose.

Modules:
- `config.py`: resolves the data directory (TIDEWICK_DATA_DIR, else ./data relative to the working directory).
- `stations.py`: loads and validates `stations.json` and `constituents/<id>.json`.
- `harmonics.py`: constituent names and speeds. Pure data plus one helper.
- `predict.py`: `height_at(station, when)` and `series(station, start, hours, step_minutes)`.
- `cli.py`: argument parsing and printing. The only module allowed to print.
- `legacy_parser.py`: reads the retired 0.x text format. Frozen; see CONTRIBUTING.md.

Decision records live in `docs/adr/`.

"""The style rules 'make check' enforces: line length, no print() outside cli.py, module docstrings."""

import ast
import sys
from pathlib import Path

MAX_LINE = 100


def main() -> int:
    problems = []
    for path in sorted(Path("tidewick").glob("*.py")):
        text = path.read_text()
        for number, line in enumerate(text.splitlines(), 1):
            if len(line) > MAX_LINE:
                problems.append(f"{path}:{number}: line longer than {MAX_LINE} characters")
        tree = ast.parse(text)
        if not ast.get_docstring(tree):
            problems.append(f"{path}: missing module docstring")
        if path.name != "cli.py":
            for node in ast.walk(tree):
                if isinstance(node, ast.Call) and getattr(node.func, "id", "") == "print":
                    problems.append(f"{path}:{node.lineno}: print() outside cli.py")
    for p in problems:
        print(p)
    print(f"{len(problems)} style problems")
    return 1 if problems else 0


if __name__ == "__main__":
    sys.exit(main())
